  <script>
    // Configuración de Corazones Flotantes
    const canvas = document.getElementById('heartsCanvas');
    const ctx = canvas.getContext('2d');
    let hearts = [];

    function resizeCanvas() {
      canvas.width = window.innerWidth;
      canvas.height = window.innerHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    class Heart {
      constructor() {
        this.reset();
      }
      reset() {
        this.x = Math.random() * canvas.width;
        this.y = canvas.height + 20;
        this.size = Math.random() * 15 + 10;
        this.speedY = Math.random() * 1.5 + 0.8;
        this.opacity = Math.random() * 0.5 + 0.3;
        this.swing = Math.random() * 0.02;
        this.swingCounter = Math.random() * Math.PI * 2;
      }
      update() {
        this.y -= this.speedY;
        this.swingCounter += this.swing;
        this.x += Math.sin(this.swingCounter) * 0.8;
        if (this.y < -20) this.reset();
      }
      draw() {
        ctx.save();
        ctx.globalAlpha = this.opacity;
        ctx.fillStyle = '#ff3366';
        ctx.beginPath();
        const x = this.x, y = this.y, s = this.size / 15;
        ctx.moveTo(x, y);
        ctx.bezierCurveTo(x - 12 * s, y - 12 * s, x - 20 * s, y + 5 * s, x, y + 20 * s);
        ctx.bezierCurveTo(x + 20 * s, y + 5 * s, x + 12 * s, y - 12 * s, x, y);
        ctx.fill();
        ctx.restore();
      }
    }

    for (let i = 0; i < 25; i++) {
      hearts.push(new Heart());
    }

    function animate() {
      ctx.clearRect(0, 0, canvas.width, canvas.height);
      hearts.forEach(h => {
        h.update();
        h.draw();
      });
      requestAnimationFrame(animate);
    }
    animate();

    // Reproductor de Música
    let isPlaying = false;
    const playBtn = document.getElementById('playBtn');

    function toggleAudio() {
      const iframe = document.getElementById('youtube-player');
      if (!isPlaying) {
        iframe.contentWindow.postMessage('{"event":"command","func":"playVideo","args":""}', '*');
        playBtn.innerText = '❚❚';
        isPlaying = true;
      } else {
        iframe.contentWindow.postMessage('{"event":"command","func":"pauseVideo","args":""}', '*');
        playBtn.innerText = '▶';
        isPlaying = false;
      }
    }

    // Intentar reproducir automáticamente al tocar la pantalla
    document.body.addEventListener('click', function() {
      if (!isPlaying) {
        toggleAudio();
      }
    }, { once: true });
  </script>
